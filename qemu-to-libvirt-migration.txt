
//example for installing solaris vm (without configuration to network...adding network right away seemed to not work for me.)
// virt install will not be exact so if starting vm via virsh start (vm-name) doesnt work, edit the xml via virsh edit (vm-name) and to shutdown the vm do virsh shutdown (vm-name)
 virt-install \
  --name solaris-vm \
  --virt-type kvm \
  --machine pc-i440fx-10.2
  --memory 8192 \
  --vcpus 4 \
  --cpu host-passthrough \
  --disk path=/home/austin/project-vm-stack/disk-drives/solaris.qcow2,bus=virtio,format=qcow2 \
  --import \
  --os-variant solaris11 \
  --network none \
  --graphics none \
  --serial pty \
  --console pty,target_type=serial \
  --check path_in_use=off

to find os variant use: virt-install --osinfo list (use grep to make it easier to find)

virsh undefine (vm-name)

to register a vm with the xml:
virsh define solaris.xml
